<script>
document.addEventListener('DOMContentLoaded', function() {
    var prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    if (prefersReduced) return;

    var bgColor = '#FFFFFF';

    document.querySelectorAll('.logos-column').forEach(function(column) {
        var track   = column.querySelector('.logos-track');
        var items   = Array.from(track.querySelectorAll('.logo-item'));
        if (!items.length) return;

        column.style.setProperty('--logos-bg', bgColor);

        items.forEach(function(item) {
            track.appendChild(item.cloneNode(true));
        });

        var isUp        = column.classList.contains('logos-column-up');
        var itemHeight  = 150 + 16;
        var totalHeight = itemHeight * items.length;
        var pos         = isUp ? 0 : -totalHeight;
        var scrolling   = true;

        setInterval(function() {
            if (!scrolling) return;
            pos += isUp ? -1.5 : 1.5;
            if (isUp && pos <= -totalHeight) pos = 0;
            if (!isUp && pos >= 0) pos = -totalHeight;
            track.style.transform = 'translateY(' + pos + 'px)';
        }, 20);

        column.addEventListener('mouseenter', function() { scrolling = false; });
        column.addEventListener('mouseleave', function() { scrolling = true; });
    });
});
</script>
