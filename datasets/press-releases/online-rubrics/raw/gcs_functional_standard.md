function createCookie(key, value, date) {
	let expiration = date
		? new Date(date).toUTCString()
		: new Date(new Date().getTime() + (365 * 24 * 60 * 60 * 1000)).toUTCString();
	let cookie = escape(key) + "=" + escape(value) + ";expires=" + expiration + "; path=/";
	document.cookie = cookie;
}

function readCookie(name) {
	let key = name + "=";
	let cookies = document.cookie.split(';');
	for (let i = 0; i < cookies.length; i++) {
		let cookie = cookies[i];
		while (cookie.charAt(0) === ' ') {
			cookie = unescape(cookie.substring(1, cookie.length));
		}
		if (cookie.indexOf(key) === 0) {
			return unescape(cookie.substring(key.length, cookie.length));
		}
	}
	return null;
}

                    <h1 class="govuk-heading-xl">Government Communication Functional Standard</h1>

                </div>

            </div>

            <div class="govuk-grid-row">

                <div class="govuk-grid-column-two-thirds">

                    
    









                    
<p>Government Communication Functional Standard GovS 011 sets the expectations for the management and practice of government communications in order to:</p>



<ul class="wp-block-list">
<li>deliver responsive and informative public service communications that support the effective delivery of HM Government policy and priorities</li>



<li>assist with the effective operation of public services</li>
</ul>



<div class="wp-block-co-custom-gutenberg-stand-out stand-out"><p><a href="https://www.gov.uk/government/publications/government-functional-standard-govs-011-communication">Government Functional Standard GovS 011: Communication </a>on GOV.UK</p></div>



<h2 class="wp-block-heading">Standard update, September 2023</h2>



<p>Version 2.1 of this standard replaces the previous edition [version 2 dated June 2021] and has the same purpose, scope and intent. </p>



<p>The main changes relate to: </p>



<ul class="wp-block-list">
<li>Minor wording changes to ensure consistency across the suite of functional standards </li>



<li>Updated references to GCS governance structure </li>



<li>Reordering to better reflect the structure of the Modern Communications Operating Model v3.0, and adding two new communications disciplines, Data &amp; Insight and Digital </li>



<li>Updated references to the GCS canon of professional guidance </li>



<li>Clarification around requirements on conduct, propriety and ethics.</li>
</ul>



<hr class="wp-block-separator has-css-opacity"/>



<h2 class="wp-block-heading">Details</h2>



<p>The standard provides direction and guidance for:</p>



<ul class="wp-block-list">
<li>permanent secretaries, directors-general (and chief executive officers (CEO) of arm’s length bodies)</li>



<li>senior leadership and board members within organisations;</li>



<li>those commissioning communications</li>



<li>communications leads within organisations</li>



<li>those involved in developing, managing and delivering communications</li>



<li>third parties engaged in government communications, including in partnership marketing</li>
</ul>
