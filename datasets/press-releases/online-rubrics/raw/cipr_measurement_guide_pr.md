    (function () {
        const dialogDownloadButton = document.querySelector('#gallery-dialog-download-button');

        dialogDownloadButton.addEventListener('click', () => {
            setTimeout(() => {
                downloadDialog.close();

        document.addEventListener('DOMContentLoaded', () => {
            downloadDialog.querySelectorAll('[data-dialog-action="cancel"]')

                    btn.addEventListener('keydown', (e) => {
                        if (e.key === 'Enter' || e.key === ' ') {
                            e.preventDefault();
                            downloadDialog.close();
                        }
                    });
                });
        });

        pp_jquery(document).on('click', '#gallery-download-all-4903230', (event) => {
            event.preventDefault();
            const btn = event.currentTarget;
            const ready = btn.dataset.ready;
            const data = {"host":"https://newsroom.cipr.co.uk/","pageType":"release","pageId":"473165","moduleId":"4903230","hash":"9fb0d4620db78528a3992f2fce19586ba2d573e6","preview":""};

            if (ready === 'error') {
                downloadDialog.querySelector('#gallery-dialog-error').classList.add('ppDialog-hidden');
                downloadDialog.querySelector('#gallery-dialog-waiting-spinner').classList.add('ppDialog-hidden');
                downloadDialog.querySelector('#gallery-dialog-waiting-wrapper').classList.remove('ppDialog-hidden');
                btn.dataset.ready = 'false';
            }

            handleDownload(data, downloadDialog, btn);
            downloadDialog.showModal();
        });


        function handleDownload(data, dialog, triggerBtn) {
            const spinnerWrapper = dialog.querySelector('#gallery-dialog-waiting-wrapper');
            const errorPara = dialog.querySelector('#gallery-dialog-error');
            spinnerWrapper.classList.remove('ppDialog-hidden');
            dialogDownloadButton.classList.add('ppDialog-hidden');

            pp_jquery.post(`${data.host}services/assetDownloadGenerator.php`, {
                module: 'gallery',
                pageType: data.pageType,
                pageId: data.pageId,
                moduleId: data.moduleId,
                hash: data.hash,
                preview: data.preview
            }, resp => {
                if (resp.hash) {
                    window.ppAssetDownloadSSE(resp.hash, {
                        success: url => {
                            spinnerWrapper.classList.add('ppDialog-hidden');
                            dialogDownloadButton.href = url;
                            dialogDownloadButton.classList.remove('ppDialog-hidden');
                            triggerBtn.dataset.ready = 'true';
                        },
                        error: err => {
                            console.error(err);
                            spinnerWrapper.classList.add('ppDialog-hidden');
                            errorPara.classList.remove('ppDialog-hidden');
                            triggerBtn.dataset.ready = 'error';
                        }
                    });
                } else {
                    spinnerWrapper.classList.add('ppDialog-hidden');
                    errorPara.classList.remove('ppDialog-hidden');
                    triggerBtn.dataset.ready = 'error';
                }
            }).fail(() => {
                spinnerWrapper.classList.add('ppDialog-hidden');
                errorPara.classList.remove('ppDialog-hidden');
                triggerBtn.dataset.ready = 'error';
            });
